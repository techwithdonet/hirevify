import { createSupabaseBrowserClient } from '@/src/lib/supabase';

export interface SkillAssessment {
  id: string;
  slug: string;
  title: string;
  description: string | null;
  category: string | null;
  level: string;
  duration_minutes: number;
  passing_score: number;
  question_type: string;
  skills: string[];
  is_active: boolean;
  created_at: string;
  updated_at: string;
}

class AssessmentsService {
  async getAssessments() {
    try {
      const supabase = createSupabaseBrowserClient();

      const { data, error } = await supabase
        .from('skill_assessments')
        .select('*')
        .eq('is_active', true)
        .order('created_at', { ascending: true });

      if (error) {
        console.warn('Assessment fetch failed:', error?.message || error);
        return { data: [], error };
      }

      return { data: data || [], error: null };
    } catch (error: any) {
      console.warn('Assessment service failed:', error?.message || error);
      return { data: [], error };
    }
  }

  async getMyResults(userId?: string) {
    try {
      const supabase = createSupabaseBrowserClient();

      const {
        data: { user },
        error: authError,
      } = await supabase.auth.getUser();

      if (authError || !user?.id) {
        console.warn('No Supabase auth user found:', authError?.message);
        return { data: [], error: authError || new Error('No auth user') };
      }

      const { data, error } = await supabase
        .from('assessment_results')
        .select(`
          id,
          user_id,
          assessment_id,
          score,
          time_spent,
          answers,
          status,
          passed,
          submitted_at,
          created_at
        `)
        .eq('user_id', user.id)
        .order('submitted_at', { ascending: false });

      if (error) {
        console.warn('Assessment results fetch failed:', {
          message: error.message,
          details: error.details,
          hint: error.hint,
          code: error.code,
        });

        return { data: [], error };
      }

      return { data: data || [], error: null };
    } catch (error: any) {
      console.warn('Assessment results service failed:', error?.message || error);
      return { data: [], error };
    }
  }

  async saveResult(input: {
    userId?: string;
    assessmentId: string;
    score: number;
    timeSpent?: number;
    answers?: Record<string, string>;
    passed?: boolean;
  }) {
    try {
      const supabase = createSupabaseBrowserClient();

      const {
        data: { user },
        error: authError,
      } = await supabase.auth.getUser();

      if (authError || !user?.id) {
        console.warn('No Supabase auth user found:', authError?.message);
        return { data: null, error: authError || new Error('No auth user') };
      }

      const { data, error } = await supabase
        .from('assessment_results')
        .insert({
          user_id: user.id,
          assessment_id: input.assessmentId,
          score: input.score,
          time_spent: input.timeSpent || 0,
          answers: input.answers || {},
          status: 'completed',
          passed: input.passed ?? input.score >= 70,
          submitted_at: new Date().toISOString(),
        })
        .select()
        .single();

      if (error) {
        console.warn('Assessment result save failed:', {
          message: error.message,
          details: error.details,
          hint: error.hint,
          code: error.code,
        });

        return { data: null, error };
      }

      return { data, error: null };
    } catch (error: any) {
      console.warn('Assessment result service failed:', error?.message || error);
      return { data: null, error };
    }
  }
}

export const assessmentsService = new AssessmentsService();
